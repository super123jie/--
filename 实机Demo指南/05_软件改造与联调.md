## 二、新增文件 1：`algorithm/src/hw_drivers.py`

在 Pi 上 `~/homecare/algorithm/src/` 目录下新建：

```python
"""硬件驱动封装层。仅在 Pi + HOMECARE_HW=1 时生效。

import 失败（如在 Windows 调试）时自动 fallback 到 mock，
不阻塞主流程。
"""
from __future__ import annotations
import os
import threading

HW_AVAILABLE = os.environ.get("HOMECARE_HW") == "1"

# 灯带状态缓存
_strip = None
_relays = {}
_servos = {}
_fan = None
_buzzer = None
_lock = threading.Lock()


def _init():
    """惰性初始化所有 GPIO。"""
    global _strip, _relays, _servos, _fan, _buzzer
    if not HW_AVAILABLE:
        return False
    if _strip is not None:
        return True
    try:
        from rpi_ws281x import PixelStrip
        from gpiozero import OutputDevice, AngularServo, Buzzer
        _strip = PixelStrip(60, 18, 800000, 10, False, 80, 0)
        _strip.begin()
        _relays = {
            "door":  OutputDevice(5,  active_high=False, initial_value=False),
            "gas":   OutputDevice(6,  active_high=False, initial_value=False),
            "tv":    OutputDevice(16, active_high=False, initial_value=False),
            "purifier": OutputDevice(20, active_high=False, initial_value=False),
        }
        _servos = {
            "curtain": AngularServo(13, min_angle=0, max_angle=180,
                                     min_pulse_width=0.0005, max_pulse_width=0.0025),
            "door":    AngularServo(19, min_angle=0, max_angle=180,
                                     min_pulse_width=0.0005, max_pulse_width=0.0025),
        }
        _buzzer = Buzzer(21)
        _fan = _relays["purifier"]  # 简化：复用净化器继电器作为风扇
        return True
    except Exception as e:
        print(f"[hw_drivers] 初始化失败: {e}; 回退 mock 模式")
        return False


# ---------- 对外 API ----------
def lights_set(brightness: int, color: tuple = (255, 180, 100)):
    """设置灯带亮度 0-100 与颜色。"""
    if not _init():
        return False
    from rpi_ws281x import Color
    with _lock:
        _strip.setBrightness(int(brightness * 255 / 100))
        for i in range(_strip.numPixels()):
            _strip.setPixelColor(i, Color(*color))
        _strip.show()
    return True


def relay_set(name: str, on: bool):
    """name ∈ door / gas / tv / purifier"""
    if not _init():
        return False
    if name not in _relays:
        return False
    if on:
        _relays[name].on()
    else:
        _relays[name].off()
    return True


def servo_angle(name: str, deg: int):
    """name ∈ curtain / door"""
    if not _init():
        return False
    if name not in _servos:
        return False
    _servos[name].angle = max(0, min(180, deg))
    return True


def fan_set(on: bool):
    if not _init():
        return False
    if on:
        _fan.on()
    else:
        _fan.off()
    return True


def buzzer_beep(ms: int = 200):
    if not _init():
        return False
    import time
    _buzzer.on()
    time.sleep(ms / 1000.0)
    _buzzer.off()
    return True


def emergency_alarm():
    """紧急告警：蜂鸣器 + 红灯闪。"""
    if not _init():
        return False
    import time
    for _ in range(3):
        lights_set(100, (255, 0, 0))
        buzzer_beep(150)
        time.sleep(0.1)
        lights_set(0, (0, 0, 0))
        time.sleep(0.1)
    return True
```

```python
@_register("control_light", ...)
def control_light(state, location, action, brightness=None):
    before = copy.deepcopy(_get_device(state, location, "light"))
    dev = _get_device(state, location, "light")
    if action == "on":
        dev["power"] = "on"
        dev.setdefault("brightness", 80)
    elif action == "off":
        dev["power"] = "off"
    elif action == "set" and brightness is not None:
        dev["power"] = "on"
        dev["brightness"] = max(0, min(100, brightness))
    # ↓↓↓ 新增 ↓↓↓
    if HW_AVAILABLE:
        if dev["power"] == "on":
            hw_drivers.lights_set(dev["brightness"], (255, 180, 100))
        else:
            hw_drivers.lights_set(0, (0, 0, 0))
    # ↑↑↑ 新增 ↑↑↑
    return {...}
```

## 四、新增文件 3：`algorithm/src/sensor_loop.py`

在 Pi 上后台线程读传感器，自动触发 Agent：

```python
"""传感器后台轮询线程：把环境事件转成 Agent 输入。"""
from __future__ import annotations
import os
import threading
import time

HW_AVAILABLE = os.environ.get("HOMECARE_HW") == "1"

_running = False
_status = {"temperature": None, "humidity": None,
            "gas": False, "person": False, "last_update": 0}


def get_status() -> dict:
    return dict(_status)


def _loop(run_fn, sid):
    if not HW_AVAILABLE:
        return
    try:
        from gpiozero import DigitalInputDevice
        import board
        import adafruit_dht
    except ImportError:
        print("[sensor_loop] 依赖缺失，停止")
        return

    dht = adafruit_dht.DHT22(board.D4)
    mq2 = DigitalInputDevice(17, pull_up=False)
    pir = DigitalInputDevice(27, pull_up=False)

    last_gas = False
    last_person = False
    last_person_time = 0
    while _running:
        try:
            _status["temperature"] = dht.temperature
            _status["humidity"] = dht.humidity
        except Exception:
            pass
        gas = (mq2.value == 0)
        person = (pir.value == 1)
        _status["gas"] = gas
        _status["person"] = person
        _status["last_update"] = time.time()

        # 边沿触发：只在状态变化时触发 Agent
        if gas and not last_gas:
            print("[sensor] 检测到燃气泄漏 → 触发 safety.gas_leak")
            run_fn("传感器检测到厨房燃气泄漏", session_id=sid)
        if person and not last_person and time.time() - last_person_time > 30:
            print("[sensor] 检测到有人 → 触发 mode.home（仅冷启动后第一次）")
            # 可选：根据门锁状态判断回家 / 离家
            last_person_time = time.time()

        last_gas = gas
        last_person = person
        time.sleep(2)


def start(run_fn, sid):
    global _running
    if _running or not HW_AVAILABLE:
        return
    _running = True
    t = threading.Thread(target=_loop, args=(run_fn, sid), daemon=True)
    t.start()


def stop():
    global _running
    _running = False
```
